# 2. Run End-to-End Pipeline on Any Query (100% Rate-Limit Proof, Scalable to 500-1000+ Papers)

python -m research_radar.pipeline --query "federated learning healthcare" --max-results 300 --corpus data/fl_papers.json --out data/output_fl

python -m research_radar.pipeline --query "graph neural networks drug discovery" --max-results 300 --corpus data/gnn_papers.json --out data/output_gnn

# 3. Fast Re-runs with Query Caching (0 API Calls, 0.00s Load Time)
# Re-run any query with different topic modeling parameters without re-fetching from S2
python -m research_radar.pipeline --query "federated learning healthcare" --max-results 300 --target-topics 6 --corpus data/fl_papers.json --out data/output_fl

# 4. Force Fresh S2 Fetch (Bypasses Local Cache)
python -m research_radar.pipeline --query "federated learning healthcare" --max-results 500 --force-fetch --corpus data/fl_500.json --out data/output_fl_500

# 5. Run Offline Sample Papers
python -m research_radar.pipeline --corpus data/sample_papers.json --out data/output

# 6. Run Standalone Visualizations (Stage 5)
