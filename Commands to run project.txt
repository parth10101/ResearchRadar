# 1. Activate Environment (PowerShell)
.\venv311\Scripts\Activate.ps1

# ======================================================================
# FULL-STACK & WEB UI EXECUTION (FRONTEND + BACKEND TERMINAL)
# ======================================================================

# OPTION A: 1-Click Unified Mode (Easiest - 1 Terminal)
# Serves both the FastAPI REST API and compiled React SPA on Port 8000:
.\venv311\Scripts\python.exe -m uvicorn research_radar.api:app --host 127.0.0.1 --port 8000 --reload
# Then open: http://127.0.0.1:8000
# (Or simply double-click 'run_app.bat')

# OPTION B: Live Development Mode (2 Terminals with Hot-Reload)
# Terminal 1 (FastAPI Backend):
.\venv311\Scripts\python.exe -m uvicorn research_radar.api:app --host 127.0.0.1 --port 8000 --reload
# Terminal 2 (Vite Frontend Hot-Reload):
cd frontend
npm run dev
# Then open: http://localhost:5173
# (Or simply double-click 'run_dev.bat')

# Rebuilding the Frontend after modifications:
cd frontend
npm run build

# 2. Run End-to-End Pipeline on Any Query (100% Rate-Limit Proof, Scalable to 500-1000+ Papers)
# Query 1: Federated Learning
python -m research_radar.pipeline --query "federated learning healthcare" --max-results 300 --corpus data/fl_papers.json --out data/output_fl

# Query 2: Reasoning & Chain-of-Thought
python -m research_radar.pipeline --query "chain of thought reasoning" --max-results 300 --corpus data/cot_papers.json --out data/output_cot

# Query 3: Graph Neural Networks in Drug Discovery
python -m research_radar.pipeline --query "graph neural networks drug discovery" --max-results 300 --corpus data/gnn_papers.json --out data/output_gnn

# 3. Fast Re-runs with Query Caching (0 API Calls, 0.00s Load Time)
# Re-run any query with different topic modeling parameters without re-fetching from S2
python -m research_radar.pipeline --query "federated learning healthcare" --max-results 300 --target-topics 6 --corpus data/fl_papers.json --out data/output_fl

# 4. Force Fresh S2 Fetch (Bypasses Local Cache)
python -m research_radar.pipeline --query "federated learning healthcare" --max-results 500 --force-fetch --corpus data/fl_500.json --out data/output_fl_500

# 5. Run Offline Sample Papers
python -m research_radar.pipeline --corpus data/sample_papers.json --out data/output

# 6. Run Standalone Stages
# Tab 02: Citations (The Epistemic Citation Journey & Publication Deep-Dive)
python -m research_radar.stage5_citations --papers data/output/papers.json --topics data/output/topic_results.json --out data/output

# Tab 01: Domain Clusters (Epistemic Archipelago & Macro Filaments)
python -m research_radar.stage6_domain_clusters_archipelago --papers data/output/papers.json --topics data/output/topic_results.json --out data/output
